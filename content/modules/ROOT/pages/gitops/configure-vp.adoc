= Configure Validated Patterns for RHOSO

== Clone the Repositories we will use in the lab

[source,bash,role=execute]
----
cd
git clone https://github.com/rhpds/osp-on-ocp-showroom.git labrepo
git clone git@github.com:validatedpatterns-sandbox/rhoso-gitops.git rhoso-gitops
cd rhoso-gitops
----

== Configure the Validated Patterns

[source,bash,role=execute]
----
cd rhoso-gitops
cd overrides
----

Configure the file values-rhoso-gitops.yaml

[source,bash,role=execute]
----
cat > values-rhoso-gitops.yaml <<EOF
---

    targetRevision: v0.3.1

          duration: 30s

          maxDuration: 3m
